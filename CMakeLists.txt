cmake_minimum_required(VERSION 3.24)
project(openUCC VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(openucc_core src/package.cpp src/metadata.cpp src/target.cpp src/ir.cpp src/compiler.cpp src/resources.cpp src/static_mesh.cpp src/animation.cpp src/skeletal_mesh.cpp src/model.cpp src/model133.cpp)
target_include_directories(openucc_core PUBLIC include)
if(MSVC)
  target_compile_options(openucc_core PRIVATE /W4 /EHsc /utf-8 "/pathmap:${CMAKE_SOURCE_DIR}=.")
else()
  target_compile_options(openucc_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(openucc src/main.cpp)
target_link_libraries(openucc PRIVATE openucc_core)
if(MSVC)
  target_compile_options(openucc PRIVATE "/pathmap:${CMAKE_SOURCE_DIR}=.")
  target_link_options(openucc PRIVATE "$<$<CONFIG:Release>:/Brepro>")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(openucc_sound_tests tests/test_sound.cpp)
  target_link_libraries(openucc_sound_tests PRIVATE openucc_core)
  add_test(NAME sound_codecs COMMAND openucc_sound_tests)
  add_executable(openucc_skeletal_tests tests/test_skeletal_mesh.cpp)
  target_link_libraries(openucc_skeletal_tests PRIVATE openucc_core)
  add_test(NAME skeletal_mesh_codecs COMMAND openucc_skeletal_tests)
  add_executable(openucc_animation_tests tests/test_animation.cpp)
  target_link_libraries(openucc_animation_tests PRIVATE openucc_core)
  add_test(NAME animation_codecs COMMAND openucc_animation_tests)
  add_executable(openucc_envelope_tests tests/test_envelope.cpp)
  target_link_libraries(openucc_envelope_tests PRIVATE openucc_core)
  add_test(NAME filename_envelope COMMAND openucc_envelope_tests "${CMAKE_CURRENT_BINARY_DIR}/MiXeD.utx")
  add_executable(openucc_tests tests/test_core.cpp)
  target_link_libraries(openucc_tests PRIVATE openucc_core)
  add_test(NAME format_and_compiler COMMAND openucc_tests)
  add_executable(openucc_portable_fixture tests/portable_fixture.cpp)
  target_link_libraries(openucc_portable_fixture PRIVATE openucc_core)
  add_executable(openucc_resource_tests tests/test_resources.cpp)
  target_link_libraries(openucc_resource_tests PRIVATE openucc_core)
  add_test(NAME resource_codecs COMMAND openucc_resource_tests)
  add_executable(openucc_model133_tests tests/test_model133.cpp)
  target_link_libraries(openucc_model133_tests PRIVATE openucc_core)
  add_test(NAME model133_codecs COMMAND openucc_model133_tests)
  add_test(NAME model_unverified_passthrough
    COMMAND "${CMAKE_COMMAND}" "-DCLI=$<TARGET_FILE:openucc>"
      "-DFIXTURE_TOOL=$<TARGET_FILE:openucc_model133_tests>"
      "-DTEST_DIR=${CMAKE_CURRENT_BINARY_DIR}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_model_unverified_cli.cmake")
endif()
install(TARGETS openucc RUNTIME DESTINATION bin)
